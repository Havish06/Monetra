import { API_BASE_URL } from "../constants/config";
import { supabase } from "../lib/supabase";
import { saveDashboardCache } from "../lib/appCache";

export type DashboardData = {
  total_account_balance: number;
  monthly_income: number;
  monthly_expenses: number;
  monthly_net: number;
  total_savings_target: number;
  total_saved: number;
  upcoming_commitments_count: number;
  accounts: any[];
  recent_transactions: any[];
  budgets: any[];
  savings_goals: any[];
  upcoming_commitments: any[];
};

async function authenticatedFetch(path: string): Promise<Response> {
  const {
    data: { session },
    error,
  } = await supabase.auth.getSession();

  if (error || !session) {
    throw new Error("Your session has expired. Please sign in again.");
  }

  const response = await fetch(`${API_BASE_URL}${path}`, {
    method: "GET",
    headers: {
      Authorization: `Bearer ${session.access_token}`,
      "Content-Type": "application/json",
    },
  });

  if (!response.ok) {
    const errorBody = await response.text();

    console.error("API ERROR:", response.status, errorBody);

    throw new Error(
      `Request failed (${response.status}): ${errorBody}`
    );
  }

  return response;
}

export async function getDashboard(
  userId: string
): Promise<DashboardData> {
  if (!userId || userId === "undefined") {
    throw new Error("Monetra profile ID is missing.");
  }

  const response = await authenticatedFetch(
    `/api/v1/users/${userId}/dashboard`
  );

  const dashboard: DashboardData = await response.json();

  await saveDashboardCache(userId, dashboard);

  return dashboard;
}


========== src\context\AuthContext.tsx ==========


import {
  createContext,
  useContext,
  useEffect,
  useState,
  ReactNode,
} from "react";

import { Session } from "@supabase/supabase-js";
import { supabase } from "../lib/supabase";
import { API_BASE_URL } from "../constants/config";
import {
  getProfileCache,
  saveProfileCache,
  clearUserCache,
} from "../lib/appCache";

export type Profile = {
  id: string;
  email: string;
  full_name: string | null;
  currency: string;
  timezone: string;
};

type ProfileUpdates = {
  full_name?: string;
  currency?: string;
  timezone?: string;
};

type AuthContextType = {
  session: Session | null;
  profile: Profile | null;
  loading: boolean;
  signUp: (email: string, password: string, fullName: string) => Promise<void>;
  signIn: (email: string, password: string) => Promise<void>;
  signOut: () => Promise<void>;
  updateProfile: (updates: ProfileUpdates) => Promise<void>;
};

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [session, setSession] = useState<Session | null>(null);
  const [profile, setProfile] = useState<Profile | null>(null);
  const [loading, setLoading] = useState(true);

  async function syncProfile(
    accessToken: string,
    authUserId: string
  ): Promise<Profile> {
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 12000);

    try {
      const response = await fetch(`${API_BASE_URL}/api/v1/profile/me`, {
        method: "POST",
        headers: {
          Authorization: `Bearer ${accessToken}`,
          "Content-Type": "application/json",
        },
        signal: controller.signal,
      });

      if (!response.ok) {
        const error = await response.json().catch(() => ({}));
        throw new Error(
          error.detail || "Could not set up your Monetra profile"
        );
      }

      const data: Profile = await response.json();

      setProfile(data);
      await saveProfileCache(authUserId, data);

      return data;
    } catch (error) {
      if (controller.signal.aborted) {
        throw new Error("Backend connection timed out.");
      }
      throw error;
    } finally {
      clearTimeout(timeout);
    }
  }

  async function establishSession(currentSession: Session) {
    setSession(currentSession);

    const cachedProfile = await getProfileCache(
      currentSession.user.id
    );

    if (cachedProfile) {
      setProfile(cachedProfile);
      setLoading(false);

      // Refresh profile without blocking the app.
      syncProfile(
        currentSession.access_token,
        currentSession.user.id
      ).catch((error) => {
        console.error("Background profile sync failed:", error);
      });

      return;
    }

    // First login: create or retrieve the backend profile.
    await syncProfile(
      currentSession.access_token,
      currentSession.user.id
    );
  }

  useEffect(() => {
    let mounted = true;

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange(
      (_event, currentSession) => {
        if (!mounted) return;

        setSession(currentSession);

        if (!currentSession) {
          setProfile(null);
          setLoading(false);
        }
      }
    );

    async function initialize() {
      try {
        const {
          data: { session: currentSession },
          error,
        } = await supabase.auth.getSession();

        if (!mounted) return;
        if (error) throw error;

        if (!currentSession) {
          setSession(null);
          setProfile(null);
          return;
        }

        const cachedProfile = await getProfileCache(
          currentSession.user.id
        );

        if (!mounted) return;

        setSession(currentSession);

        if (cachedProfile) {
          setProfile(cachedProfile);
          setLoading(false);

          syncProfile(
            currentSession.access_token,
            currentSession.user.id
          ).catch((error) => {
            console.error("Background profile sync failed:", error);
          });
        } else {
          await syncProfile(
            currentSession.access_token,
            currentSession.user.id
          );
        }
      } catch (error) {
        console.error("Profile setup failed:", error);
      } finally {
        if (mounted) setLoading(false);
      }
    }

    initialize();

    return () => {
      mounted = false;
      subscription.unsubscribe();
    };
  }, []);

  async function signUp(
    email: string,
    password: string,
    fullName: string
  ) {
    setLoading(true);

    try {
      const { data, error } = await supabase.auth.signUp({
        email,
        password,
        options: {
          data: { full_name: fullName },
        },
      });

      if (error) throw error;

      if (data.session) {
        await establishSession(data.session);
      }
    } finally {
      setLoading(false);
    }
  }

  async function signIn(email: string, password: string) {
    setLoading(true);

    try {
      const { data, error } = await supabase.auth.signInWithPassword({
        email,
        password,
      });

      if (error) throw error;

      if (data.session) {
        await establishSession(data.session);
      }
    } finally {
      setLoading(false);
    }
  }

  async function updateProfile(updates: ProfileUpdates) {
    if (!session) {
      throw new Error("Please sign in again.");
    }

    const response = await fetch(`${API_BASE_URL}/api/v1/profile/me`, {
      method: "PATCH",
      headers: {
        Authorization: `Bearer ${session.access_token}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify(updates),
    });

    if (!response.ok) {
      const error = await response.json().catch(() => ({}));
      throw new Error(error.detail || "Could not update your profile");
    }

    const updatedProfile: Profile = await response.json();

    setProfile(updatedProfile);
    await saveProfileCache(session.user.id, updatedProfile);
  }

  async function signOut() {
    const authUserId = session?.user.id;

    const { error } = await supabase.auth.signOut();

    if (error) throw error;

    if (authUserId) {
      await clearUserCache(authUserId);
    }

    setProfile(null);
    setSession(null);
  }

  return (
    <AuthContext.Provider
      value={{
        session,
        profile,
        loading,
        signUp,
        signIn,
        signOut,
        updateProfile,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);

  if (!context) {
    throw new Error("useAuth must be used inside AuthProvider");
  }

  return context;
}

export default function HomeScreen() {
  const { profile, loading: authLoading } = useAuth();

  const [data, setData] = useState<DashboardData | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [retry, setRetry] = useState(0);

  const router = useRouter();

  useEffect(() => {
    if (authLoading) return;

    if (!profile?.id) {
      setLoading(false);
      setData(null);
      setError("Your Monetra profile is unavailable. Please sign in again.");
      return;
    }

    let active = true;

    async function initializeDashboard() {
      setError(null);

      // Show the loading screen only when no data is available.
      if (!data) {
        setLoading(true);
      }

      try {
        // 1. Read cached dashboard first.
        const cached = await getDashboardCache(profile!.id);

        if (!active) return;

        // 2. Display cached data immediately.
        if (cached) {
          setData(cached);
          setLoading(false);
        }

        // 3. Fetch fresh data from the backend.
        const fresh = await getDashboard(profile!.id);

        if (!active) return;

        // 4. Replace cached data with fresh data.
        setData(fresh);
        setError(null);
      } catch (err: unknown) {
        if (!active) return;

        const message =
          err instanceof Error
            ? err.message
            : "Unable to load your dashboard.";

        setError(message);

        // Keep cached data visible if the request fails.
        console.error("Dashboard refresh failed:", message);
      } finally {
        if (active) {
          setLoading(false);
        }
      }
    }

    initializeDashboard();

    return () => {
      active = false;
    };
  }, [authLoading, profile?.id, retry]);

  if (authLoading || (loading && !data)) {
    return (
      <Screen>
        <View style={styles.state}>
          <ActivityIndicator
            size="large"
            color={theme.colors.primary}
          />

          <Text style={styles.stateText}>
            Loading your Monetra dashboard...
          </Text>
        </View>
      </Screen>
    );
  }

  if (error && !data) {
    return (
      <Screen>
        <View style={styles.state}>
          <Ionicons
            name="cloud-offline-outline"
            size={36}
            color={theme.colors.warning}
          />

          <Text style={styles.stateTitle}>
            Couldn't load your dashboard
          </Text>

          <Text style={styles.stateText}>
            {error}
          </Text>

          <Pressable
            style={styles.retryButton}
            onPress={() => setRetry((value) => value + 1)}
          >
            <Text style={styles.retryText}>Try again</Text>
          </Pressable>
        </View>
      </Screen>
    );
  }

  if (!data) return null;

  const firstName =
    profile?.full_name?.trim().split(/\s+/)[0] || "there";

  return (
    <Screen>
      <ScrollView
        showsVerticalScrollIndicator={false}
        contentContainerStyle={styles.content}
      >
        <View style={styles.header}>
          <View>
            <Text style={styles.eyebrow}>
              YOUR FINANCIAL OVERVIEW
            </Text>

            <Text style={styles.greeting}>
              Welcome, {firstName}.
            </Text>
          </View>

          <Pressable
            style={styles.notification}
            onPress={() => router.push("/profile")}
          >
            <Ionicons
              name="person-circle-outline"
              size={23}
              color={theme.colors.text}
            />
          </Pressable>
        </View>

        <Text style={styles.demoLabel}>
          LIVE DATA · YOUR MONETRA ACCOUNTS
        </Text>

        <BalanceCard
          available={data.total_account_balance}
          income={data.monthly_income}
          spent={data.monthly_expenses}
        />

        <View style={styles.insight}>
          <View style={styles.insightIcon}>
            <Ionicons
              name="sparkles"
              size={17}
              color={theme.colors.primary}
            />
          </View>

          <View style={styles.insightContent}>
            <Text style={styles.insightLabel}>
              MONETRA OVERVIEW
            </Text>

            <Text style={styles.insightText}>
              Your monthly net cash flow is{" "}
              {formatCurrency(data.monthly_net)}. You have{" "}
              {data.upcoming_commitments_count} upcoming commitments
              to keep in mind.
            </Text>
          </View>
        </View>

        <View style={styles.section}>
          <SectionHeader title="Savings progress" />

          <View style={styles.statCard}>
            <View style={styles.statRow}>
              <Text style={styles.statLabel}>Total saved</Text>

              <Text style={styles.statValue}>
                {formatCurrency(data.total_saved)}
              </Text>
            </View>

            <View style={styles.statRow}>
              <Text style={styles.statLabel}>Savings target</Text>

              <Text style={styles.statValue}>
                {formatCurrency(data.total_savings_target)}
              </Text>
            </View>
          </View>
        </View>

        <View style={styles.section}>
          <SectionHeader title="Upcoming commitments" />

          {data.upcoming_commitments.length === 0 ? (
            <Text style={styles.emptyText}>
              No upcoming commitments. You're all clear!
            </Text>

  return new Intl.NumberFormat("en-IN", {
    style: "currency",
    currency: "INR",
    maximumFractionDigits: 0,
  }).format(Number(value) || 0);
}

import AsyncStorage from "@react-native-async-storage/async-storage";
import type { Profile } from "../context/AuthContext";
import type { DashboardData } from "../services/api";

const profileKey = (userId: string) => `monetra_profile_${userId}`;
const dashboardKey = (userId: string) => `monetra_dashboard_${userId}`;

export async function saveProfileCache(
  userId: string,
  profile: Profile
) {
  try {
    await AsyncStorage.setItem(
      profileKey(userId),
      JSON.stringify(profile)
    );
  } catch (error) {
    console.warn("Profile cache save failed:", error);
  }
}

export async function getProfileCache(
  userId: string
): Promise<Profile | null> {
  try {
    const value = await AsyncStorage.getItem(profileKey(userId));
    return value ? JSON.parse(value) : null;
  } catch {
    return null;
  }
}

export async function saveDashboardCache(
  userId: string,
  dashboard: DashboardData
) {
  try {
    await AsyncStorage.setItem(
      dashboardKey(userId),
      JSON.stringify(dashboard)
    );
  } catch (error) {
    console.warn("Dashboard cache save failed:", error);
  }
}

export async function getDashboardCache(
  userId: string
): Promise<DashboardData | null> {

def calculate_spent(
    user_id: uuid.UUID,
    budget: Budget,
    db: Session,
) -> Decimal:
    query = db.query(
        func.coalesce(func.sum(Transaction.amount), 0)
    ).filter(
        Transaction.user_id == user_id,
        Transaction.transaction_type == "expense",
        Transaction.category == budget.category,
        Transaction.transaction_date >= budget.start_date,
    )

    if budget.end_date:
        query = query.filter(
            Transaction.transaction_date < (
                budget.end_date + timedelta(days=1)
            )
        )

    return Decimal(str(query.scalar()))


def make_summary(
    user_id: uuid.UUID,
    budget: Budget,
    db: Session,
) -> BudgetSummary:
    spent = calculate_spent(user_id, budget, db)
    remaining = budget.limit_amount - spent

    utilization = (
        spent / budget.limit_amount * Decimal("100")
    ).quantize(Decimal("0.01"))

    return BudgetSummary(
        budget=budget,
        spent=spent,
        remaining=remaining,
        utilization_percentage=utilization,
        is_exceeded=spent > budget.limit_amount,
    )

from app.models.user import User
from app.models.savings_goal import SavingsGoal

def verify_user(user_id: uuid.UUID, db: Session):
    if not db.get(User, user_id):
        raise HTTPException(status_code=404, detail="User not found")


def calculate_summary(
    user_id: uuid.UUID,
    goal: SavingsGoal,
    db: Session,
) -> SavingsGoalSummary:
    total = db.query(
        func.coalesce(func.sum(SavingsContribution.amount), 0)
    ).filter(
        SavingsContribution.goal_id == goal.id,
        SavingsContribution.user_id == user_id,
    ).scalar()

    total_saved = Decimal(str(total))
    remaining = max(goal.target_amount - total_saved, Decimal("0"))

    progress = min(
        total_saved / goal.target_amount * Decimal("100"),
        Decimal("100"),
    ).quantize(Decimal("0.01"))

    return SavingsGoalSummary(
        goal=goal,
        total_saved=total_saved,
        remaining=remaining,
        progress_percentage=progress,
        is_completed=total_saved >= goal.target_amount,
    )

import uuid
from datetime import datetime, date, time, timedelta
from decimal import Decimal
from zoneinfo import ZoneInfo

from fastapi import APIRouter, Depends, HTTPException

def money(value):
    return Decimal(str(value or 0))


@router.get("", response_model=DashboardResponse)
def get_dashboard(
    user_id: uuid.UUID,
    db: Session = Depends(get_db),
):
    user = db.query(User).filter(User.id == user_id).first()

    if not user:
        raise HTTPException(status_code=404, detail="User not found")

    # Use the user's configured timezone for dashboard date boundaries.
    try:
        user_timezone = ZoneInfo(user.timezone)
    except Exception:
        user_timezone = ZoneInfo("UTC")

    today = datetime.now(user_timezone).date()
    tomorrow = today + timedelta(days=1)
    next_30_days = today + timedelta(days=30)

    month_start = date(today.year, today.month, 1)

    if today.month == 12:
        next_month_start = date(today.year + 1, 1, 1)
    else:
        next_month_start = date(today.year, today.month + 1, 1)

    month_start_dt = datetime.combine(
        month_start, time.min, tzinfo=user_timezone
    )
    next_month_dt = datetime.combine(
        next_month_start, time.min, tzinfo=user_timezone
    )
    tomorrow_dt = datetime.combine(
        tomorrow, time.min, tzinfo=user_timezone
    )

    # 1. Financial accounts
    accounts = (
        db.query(FinancialAccount)
        .filter(FinancialAccount.user_id == user_id)
        .order_by(FinancialAccount.created_at.desc())
        .all()
    )

    total_account_balance = sum(
        (money(account.balance) for account in accounts),
        Decimal("0"),
    )

    # 2. Current month's income and expenses
    monthly_income = money(
        db.query(func.coalesce(func.sum(Transaction.amount), 0))
        .filter(
            Transaction.user_id == user_id,
            Transaction.transaction_type == "income",
            Transaction.transaction_date >= month_start_dt,
            Transaction.transaction_date < next_month_dt,
        )
        .scalar()
    )

    monthly_expenses = money(
        db.query(func.coalesce(func.sum(Transaction.amount), 0))
        .filter(
            Transaction.user_id == user_id,
            Transaction.transaction_type == "expense",
            Transaction.transaction_date >= month_start_dt,
            Transaction.transaction_date < next_month_dt,
        )
        .scalar()
    )

    # 3. Recent transactions
    recent_transactions = (
        db.query(Transaction)
        .filter(Transaction.user_id == user_id)
        .order_by(Transaction.transaction_date.desc())
        .limit(5)
        .all()
    )

    # 4. Active budgets and their current spending
    active_budgets = (
        db.query(Budget)
        .filter(
            Budget.user_id == user_id,
            Budget.start_date <= today,
            (Budget.end_date.is_(None) | (Budget.end_date >= today)),
        )
        .all()
    )

    budget_overviews = []

    for budget in active_budgets:
        spent = money(
            db.query(func.coalesce(func.sum(Transaction.amount), 0))
            .filter(
                Transaction.user_id == user_id,
                Transaction.transaction_type == "expense",
                Transaction.category == budget.category,
                Transaction.transaction_date >= datetime.combine(
                    budget.start_date, time.min, tzinfo=user_timezone
                ),
                Transaction.transaction_date < tomorrow_dt,
            )
            .scalar()
        )

        limit_amount = money(budget.limit_amount)
        remaining = max(limit_amount - spent, Decimal("0"))

        percentage = (
            (spent / limit_amount) * 100
            if limit_amount > 0
            else Decimal("0")
        )

        budget_overviews.append({
            "id": budget.id,
            "name": budget.name,
            "category": budget.category,
            "limit_amount": limit_amount,
            "spent": spent,
            "remaining": remaining,
            "utilization_percentage": percentage.quantize(
                Decimal("0.01")
            ),
        })

    # 5. Savings goals and contributions
    active_goals = (
        db.query(SavingsGoal)
        .filter(
            SavingsGoal.user_id == user_id,
            SavingsGoal.status == "active",
        )
        .all()
    )

    savings_goal_overviews = []
    total_savings_target = Decimal("0")
    total_saved = Decimal("0")

    for goal in active_goals:
        saved = money(
            db.query(func.coalesce(func.sum(SavingsContribution.amount), 0))
            .filter(
                SavingsContribution.goal_id == goal.id,
                SavingsContribution.user_id == user_id,
            )
            .scalar()
        )

        target = money(goal.target_amount)
        remaining = max(target - saved, Decimal("0"))

        percentage = (
            min((saved / target) * 100, Decimal("100"))
            if target > 0
            else Decimal("0")
        )

        total_savings_target += target
        total_saved += saved

        savings_goal_overviews.append({
            "id": goal.id,
            "name": goal.name,
            "target_amount": target,
            "total_saved": saved,
            "remaining": remaining,
            "progress_percentage": percentage.quantize(
                Decimal("0.01")
            ),
            "target_date": goal.target_date,
            "status": goal.status,
        })

    # 6. Upcoming commitments in the next 30 days
    upcoming_commitments = (
        db.query(Commitment)
        .filter(
            Commitment.user_id == user_id,
            Commitment.status == "active",
            Commitment.next_due_date.isnot(None),
            Commitment.next_due_date >= today,
            Commitment.next_due_date <= next_30_days,
        )
        .order_by(Commitment.next_due_date.asc())
        .all()
    )

    return {
        "total_account_balance": total_account_balance,
        "monthly_income": monthly_income,
        "monthly_expenses": monthly_expenses,
        "monthly_net": monthly_income - monthly_expenses,
        "total_savings_target": total_savings_target,
        "total_saved": total_saved,
        "upcoming_commitments_count": len(upcoming_commitments),
        "accounts": accounts,
        "recent_transactions": recent_transactions,
        "budgets": budget_overviews,
        "savings_goals": savings_goal_overviews,
        "upcoming_commitments": upcoming_commitments,
    }

import uuid

from sqlalchemy import (
    String,
    Uuid,
    DateTime,
    Numeric,
    ForeignKey,
    func,
)
from sqlalchemy.orm import Mapped, mapped_column

from app.db.database import Base


class Transaction(Base):
    __tablename__ = "transactions"

    id: Mapped[uuid.UUID] = mapped_column(
        Uuid,
        primary_key=True,
        default=uuid.uuid4,
    )

    user_id: Mapped[uuid.UUID] = mapped_column(
        Uuid,
        ForeignKey("users.id", ondelete="CASCADE"),
        nullable=False,
        index=True,
    )

    account_id: Mapped[uuid.UUID | None] = mapped_column(
        Uuid,
        ForeignKey("financial_accounts.id", ondelete="SET NULL"),
        nullable=True,
        index=True,
    )

    title: Mapped[str] = mapped_column(
        String(150),
        nullable=False,
    )

    amount: Mapped[Decimal] = mapped_column(
        Numeric(14, 2),
        nullable=False,
    )

    transaction_type: Mapped[str] = mapped_column(
        String(20),
        nullable=False,
    )

    category: Mapped[str | None] = mapped_column(
        String(50),
        nullable=True,
    )

    description: Mapped[str | None] = mapped_column(
        String(500),
        nullable=True,
    )

    transaction_date: Mapped[DateTime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
    )

    created_at: Mapped[DateTime] = mapped_column(
        DateTime(timezone=True),
        server_default=func.now(),
        nullable=False,
    )
